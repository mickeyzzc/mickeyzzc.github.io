gb28181-go 教程：从设备注册到平台收流

September 30, 2026 MiBeeOSS Go, GB28181, 监控摄像头, 开源 MiBee 开源项目实践系列 5976 字 12 分钟阅读
🔊

gb28181-go 是五个协议库里唯一「一册全览」的：同一个 module 里既有设备端（UAC，把一台媒体源注册到国标平台），也有平台端（UAS，收设备的注册、向它要流）。设备端代码来自 MiBee Eye 树莓派相机固件，平台端从 MiBeeNvr 的国标接入实现整体抽出，两头每天都在生产环境里对着咬。

两个角色可以分开用——做相机固件的人用 device/，做平台的人用 platform/+platform/sip/，互不牵连。这篇按「能直接抄去用」的标准写：每个角色从建工程讲到完整程序，每份配置逐字段说明，读完就能在自己项目里落地。

要求 Go 1.26 以上（库强制 1.26.6 工具链，为拉 stdlib 安全修复）。依赖有几个，第一次 go get 自动带下来：emmansun/gmsm（GB 35114 的 SM 算法）、ghettovoice/gosip（只有平台端 SIP 用，设备端是手写 SIP）、pion/rtp、golang.org/x/text（GBK 字符集）。文档在 MiBee 协议库手册。

对练伙伴：设备端要一台国标平台当对手，开源的 WVP 就能对练；平台端反过来，随便找一台支持 GB28181 的摄像头，或者用本库设备端当假相机。

package main

import (
	"context"
	"log"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/mickeyzzc/gb28181-go/device"
)

func main() {
	cfg := device.Config{
		Enabled:               true,
		PlatformSIPAddress:    "192.168.1.10",
		PlatformSIPPort:       5060,
		DeviceID:              "34020000001320000001", // 20 位国标编码
		ChannelID:             "34020000001310000001",
		SIPDomain:             "3402000000",
		Password:              "your-platform-password", // 与平台约定的 digest 口令
		LocalSIPPort:          5060,
		RegisterIntervalSecs:  3600,
		HeartbeatIntervalSecs: 60,
		HeartbeatTimeoutCount: 3,
		Transport:             "udp", // udp | tcp | tls (SIPS)
	}

	// FrameHub 是现成的 FrameSource：有界通道、写满即丢
	hub := device.NewFrameHub()

	srv := device.New(cfg, device.DeviceInfo{
		Name:         "Front Door",
		Manufacturer: "MiBee Studio",
		Model:        "MiBee Eye",
	}, hub)

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	// 采集循环：换成你的编码器输出。这里用合成帧演示形状。
	go func() {
		for i := 0; ; i++ {
			au := device.AccessUnit{Timestamp: time.Now(), KeyFrame: i%25 == 0}
			if au.KeyFrame {
				au.NALUs = []device.NALU{
					{Type: 7, IsSPS: true, Data: []byte{0x67, 0x42}}, // SPS
					{Type: 8, IsPPS: true, Data: []byte{0x68, 0xEB}}, // PPS
					{Type: 5, IsIDR: true, Data: []byte{0x65, 0x01}}, // IDR
				}
			} else {
				au.NALUs = []device.NALU{{Type: 1, Data: []byte{0x41, 0x02}}}
			}
			hub.Write(au)
			time.Sleep(40 * time.Millisecond) // 25 fps
		}
	}()

	errCh := make(chan error, 1)
	go func() { errCh <- srv.Start(ctx) }() // 阻塞运行：注册、保活、目录、INVITE 全在里面

	<-ctx.Done()
	_ = srv.Deregister(context.Background()) // REGISTER Expires: 0，走完整 401 流程下线
	srv.Stop()
	if err := <-errCh; err != nil {
		log.Fatal(err)
	}
}

逐段说：

  • DeviceID 和 ChannelID 是 20 位国标编码，3402000000132 开头是网络摄像机类型的常见取法，SIPDomain 通常是编码前十位。配置错了不用等上线：Start 先跑 Validate() 快速失败，20 位位数不对当场报错。
  • device.NewFrameHub() 白送一个 FrameSource 实现：有界通道（订阅者缓冲默认 64）、写满即丢，编码器只管 Write，慢消费者丢帧不阻塞采集。要更大队列用 device.NewFrameHubWithSize(size)。
  • srv.Start(ctx) 在 UDP 模式下阻塞——注册、心跳、目录应答、INVITE 处理全部包进内部的收包循环，ctx 取消时返回。停止用 srv.Stop()，要体面下线用 srv.Deregister(ctx)（REGISTER Expires: 0 的完整 401 流程）。
  • 喂帧的 AccessUnit 里，NALU.Data 是不带 Annex-B 起始码的裸 NAL 载荷，IsSPS/IsPPS/IsIDR 三个标志位决定封装时怎么打包，关键字要如实标——它影响 RTP 时间戳和 IDR 处理。

// device.FrameSource 接口
// Subscribe(ctx context.Context) *FrameSubscription
// Unsubscribe(id string)

type captureBus struct {
	mu     sync.Mutex
	nextID int
	subs   map[string]chan device.AccessUnit
}

func newCaptureBus() *captureBus {
	return &captureBus{subs: make(map[string]chan device.AccessUnit)}
}

func (b *captureBus) Subscribe(ctx context.Context) *device.FrameSubscription {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.nextID++
	id := fmt.Sprintf("sub-%d", b.nextID)
	ch := make(chan device.AccessUnit, 64) // 有界：满即丢，不阻塞采集
	b.subs[id] = ch
	return &device.FrameSubscription{ID: id, Channel: ch}
}

func (b *captureBus) Unsubscribe(id string) {
	b.mu.Lock()
	defer b.mu.Unlock()
	if ch, ok := b.subs[id]; ok {
		close(ch)
		delete(b.subs, id)
	}
}

// 采集侧广播帧：
func (b *captureBus) broadcast(au device.AccessUnit) {
	b.mu.Lock()
	defer b.mu.Unlock()
	for _, ch := range b.subs {
		select {
		case ch <- au:
		default: // 满了就丢
		}
	}
}

要点只有一条：订阅通道必须有界、满了必须丢。INVITE 到来时库订阅一次、BYE 时退订，编码器的节奏永远不该被网络消费者的速度拖住。

帧类型是 AccessUnit{NALUs []NALU; Timestamp time.Time; KeyFrame bool}，NALU{Type byte; Data []byte; IsIDR, IsSPS, IsPPS bool}。H.264 的常识直接适用：SPS/PPS 要跟 IDR 一起喂（解码器才能初始化），关键帧标志影响平台侧的起播等待和录像切片。

    class REG,CH,RE,KA net

digest 认证（MD5 与 SHA-256、qop=auth）全自动，Password 就是和平台约定好的那串。两个值得预先知道的行为：初始 REGISTER 失败不会杀进程，设备进入 listen-only 模式按注册周期重试，网络闪断后自动恢复；注册周期到点会带鉴权重发注册，不需要干预。

生命周期相关的方法一共五个：Start(ctx)（阻塞）、Stop()、Deregister(ctx)（体面下线）、SIPPort()/SIPTCPPort()（查询实际绑定端口，LocalSIPPort 填 0 时有用）。观察平台侧时钟用 srv.PlatformDateUnix()（REGISTER 应答的 SIP-Date 头，只观察，设置时钟是宿主自己的事）。

package main

import (
	"context"
	"log"
	"time"

	"github.com/mickeyzzc/gb28181-go/gbsip"
	"github.com/mickeyzzc/gb28181-go/platform"
)

func main() {
	cfg := gbsip.Config{
		Enabled:        true,
		SIPListen:      ":5060",
		ServerID:       "34020000002000000001",
		Realm:          "3402000000",
		Password:       "your-platform-password",
		PortRange:      "30000-30050",
		MediaTransport: "udp",
	}

	dm := platform.NewDeviceManager(60 * time.Second) // 保活存活窗口
	bus := gbsip.NewEventBus(64)
	alarmCh := make(chan gbsip.Event, 8)
	_ = bus.Subscribe(gbsip.TopicGB28181Alarm, alarmCh, 0) // 报警事件流出这里

	ports := platform.NewPortManager(30000, 30050)           // RTP 端口池
	sessions := platform.NewSessionManager(ports, cfg.ServerID)
	srv := gbsip.NewServer(cfg, dm, sessions, myDeviceStore) // DeviceStore 自己实现或抄示例

	if err := srv.Start(context.Background()); err != nil {
		log.Fatalf("platform server: %v", err)
	}
	defer func() { _ = srv.Stop() }()

	// 设备上线后向一个通道要流（两步）：
	// _ = srv.InviteChannel(deviceID, channelID) // 幂等：已有会话不重复发
	// hub := sessions.GetHub(channelID)
	// _ = hub.Subscribe("recorder", func(pts int64, au [][]byte, isIDR bool) {
	//     // au 是解复用后的 NALU 载荷，isIDR 标记关键帧
	// })
	// 停流：_ = srv.ByeChannel(deviceID, channelID)

	select {} // 示例常驻；实际按你的服务框架挂接
}

五件套各管一段：DeviceManager 管设备注册表和保活存活判定；EventBus 往外发事件（TopicGB28181Alarm、TopicGB28181SnapshotFinished 两个主题）；PortManager 管 RTP 端口池；SessionManager 管会话和媒体枢纽；gbsip.NewServer 把它们接成完整的 SIP 服务器。DeviceStore 是持久化接缝，八个方法（设备/通道的增删查、离线标记、通道绑相机），内存版和 SQLite 版都在示例里。

取流的完整数据通路：InviteChannel 发 INVITE（SDP 协商在库内），平台的 NewReceiver 在端口池里收 RTP、解 RTP/PS、抖动缓冲、SSRC 锁定，解出的访问单元经 GetHub 拿到的 FrameHub 回调扇出。回调签名 func(pts int64, au [][]byte, isIDR bool)——pts 毫秒时间戳，au 是解好的 NALU 载荷组，直接接你的录像或 AI 管线。慢消费者不影响别人：hub.Dropped() 有丢帧计数。

2022 版、级联与 GB 35114

  • GB/T 28181-2022：给 Config.ProtocolVersion 设 "3.0"（2016 对应 "2.0"），REGISTER 就带上 X-GB-Ver 头做版本协商，平台回的版本从 PlatformProtocolVersion() 读；不设置则线格式与 2016 完全一致。
  • 级联：platform/cascade/ 让本级作为下级向上级注册（Upstreams 支持多上级），聚合目录上报、INVITE 转发、录像段回放转发，含多级环路预防和 2022 附录 H.3 的 X-RoutePath。
  • GB 35114 A 级：build tag -tags gb35114 启用（默认构建零 SM 依赖），SM2 证书双向认证注册加带密 SM3 的 Note 头完整性，设备端平台端都有。

另外 psmux/ 是独立的 PS/RTP 复用器（H.264/H.265、G.711/AAC、超 60KB 访问单元自动切分），想做自己的推流管道可以单独用：

生产项目里的用法

这个库的两端各有明确的生产宿主，README 的包清单里 Origin 一列写的就是出处：平台端全家（platform/、platform/sip/、psmux/、manscdp/、platform/cascade/、nalutil/）从 MiBeeNvr 的国标接入实现整体抽出；设备端 device/ 来自 MiBee Eye 的 Go 相机固件（现役实现是 mibee-eye-go）。也就是说这两个项目就是库的「第一用户」，库里的互操作修复起点都是它们报出的真机问题。按功能对照：

功能生产项目用法
设备端注册 / 保活 / 目录应答mibee-eye-goLinux 板子的 UVC 相机注册到国标平台，心跳保活，答 Catalog/DeviceInfo
FrameSource 喂帧 / INVITE 推流mibee-eye-go编码输出经 FrameHub 进库，INVITE 到来自动推 RTP/PS，BYE 自动停
平台端五件套 / INVITE 收流 / RTP 接收MiBeeNvr收国标设备注册、向设备要流，解复用后的帧进它的录像与直播管线
PTZ / 录像回放 / 对讲下发MiBeeNvr平台侧全套控制面，RecordingIndex 对接它的录像库做回放
级联MiBeeNvr作为下级注册到上级平台，聚合目录、转发 INVITE
psmux / manscdp两端共用MiBeeNvr 推流侧与设备端封包共用同一套 PS/RTP 复用与 MANSCDP 编解码

看完整实现按角色选：做相机看 mibee-eye-go——纯 Go 零 CGO，UVC 采集直出，device/ 的接缝（FrameHub 喂帧、控制回调、录像索引）在真实产品里全部有对应代码；做平台看 MiBeeNvr——单二进制 NVR，它的国标接入从注册表到级联就是 platform/ 一族的本体。两端对练时拿 WVP 或 MiBeeNvr 当平台、拿 mibee-eye-go 当设备，都可以直接开跑。

现象原因处理
